export function debounce<
  TArgs extends unknown[],
  TReturn
>(
  func: (...args: TArgs) => TReturn,
  delay: number
): (...args: TArgs) => void {
  let timeoutId: NodeJS.Timeout | null = null;

**Severity**: MEDIUM | **Effort**: SMALL | **Category**: Type Safety & Security

Recursive sanitization function uses `any` without generic type preservation:

```typescript

  if (typeof data === 'string') {
    return DOMPurify.sanitize(data, {

  }

    for (const key in data) {
      if (Object.prototype.hasOwnProperty.call(data, key)) {

```

**Problem**:
- Loss of type information after sanitization
- Can't verify sanitized data matches expected schema
- Harder to maintain type safety in validation pipeline

**Recommendation**:
Use generic type preservation:

```typescript
/**
 * Recursively sanitize input while preserving type structure
 * Note: This provides type-level guarantee, not runtime validation
 */
function sanitizeInput<T>(data: T): T {
  // Handle null/undefined
  if (data == null) {
    return data;
  }

  // Sanitize strings
  if (typeof data === 'string') {
    return DOMPurify.sanitize(data, {
      ALLOWED_TAGS: [],
      ALLOWED_ATTR: [],
    }).trim() as T;
  }

  // Sanitize arrays
  if (Array.isArray(data)) {
    return data.map(item => sanitizeInput(item)) as T;
  }

  // Sanitize objects
  if (typeof data === 'object') {
    const sanitized: Record<string, unknown> = {};
    for (const key in data) {
      if (Object.prototype.hasOwnProperty.call(data, key)) {
        sanitized[key] = sanitizeInput(data[key]);
      }
    }
    return sanitized as T;
  }

  // Return primitives unchanged
  return data;
}

// Usage maintains type safety:
interface LoginInput {
  email: string;
  password: string;
}

const sanitized = sanitizeInput<LoginInput>(req.body);
// TypeScript knows sanitized has shape { email: string, password: string }
```

**Note**: This is type-level preservation only. Runtime validation is still done by Zod schemas, which is correct.
